\section{Modelado Formal de Constraint Programming}
\label{sec:cp_modeling}

El acertijo l\'ogico KenKen de orden $n$ se formula rigurosamente como un Problema de Satisfacci\'on de Restricciones (CSP) definido por la tupla $\mathcal{P} = \langle X, D, C \rangle$, donde $X$ es el conjunto de variables de decisi\'on, $D$ representa sus dominios asociados y $C$ es el conjunto de restricciones can\'onicas, globales y reificadas.

\subsection{Variables de Decisi\'on y Dominios}
Para una cuadr\'{\i}cula de dimensiones $n \times n$, se define una matriz de variables enteras:
\begin{equation}
    X = \{x_{i,j} \mid 1 \le i \le n, \; 1 \le j \le n\}
\end{equation}
donde cada variable $x_{i,j}$ denota el d\'{\i}gito asignado a la celda ubicada en la fila $i$ y columna $j$. El dominio uniforme para cada variable est\'a acotado por:
\begin{equation}
    D(x_{i,j}) = \{1, 2, \dots, n\}, \quad \forall x_{i,j} \in X
\end{equation}

\subsection{Restricciones Globales de Tablero (Cuadrado Latino)}
Para satisfacer la regla fundamental de no repetici\'on en l\'{\i}neas ortogonales, se imponen restricciones globales de desigualdad sobre cada fila y cada columna utilizando el algoritmo de filtrado de R\'egin~\cite{regin1994filtering}:
\begin{align}
    \text{AllDifferent}(\{x_{i,1}, x_{i,2}, \dots, x_{i,n}\}), & \quad \forall i \in \{1, \dots, n\} \label{eq:alldiff_row} \\
    \text{AllDifferent}(\{x_{1,j}, x_{2,j}, \dots, x_{n,j}\}), & \quad \forall j \in \{1, \dots, n\} \label{eq:alldiff_col}
\end{align}

Adicionalmente, se analiza la incorporaci\'on de restricciones globales redundantes (implicadas) basadas en la conservaci\'on de la suma triangular:
\begin{equation}
    \sum_{j=1}^n x_{i,j} = \frac{n(n+1)}{2}, \quad \sum_{i=1}^n x_{i,j} = \frac{n(n+1)}{2}
    \label{eq:redundant_sum}
\end{equation}

\subsection{Modelado de Jaulas: Variante A (Aritm\'etica Intensional)}
Sea una jaula $c = \langle V_c, T_c, \text{op}_c \rangle$, donde $V_c \subseteq X$ es el subconjunto ordenado de celdas que la componen, $T_c \in \mathbb{N}$ es el valor objetivo y $\text{op}_c \in \{=, +, -, \times, \div\}$. La formulaci\'on intensional descompone las operaciones locales seg\'un:

\begin{enumerate}
    \item \textbf{Identidad ($=$):} Para $|V_c| = 1$:
    \begin{equation}
        v_1 = T_c, \quad v_1 \in V_c
    \end{equation}

    \item \textbf{Suma ($+$):} Restricci\'on lineal global sobre el vector de variables de la jaula:
    \begin{equation}
        \sum_{v \in V_c} v = T_c
    \end{equation}

    \item \textbf{Multiplicaci\'on ($\times$):} Descomposici\'on encadenada mediante variables auxiliares de producto $p_k \in [1, n^{k+1}]$ implementada v\'{\i}a \texttt{AddMultiplicationEquality}:
    \begin{align}
        p_1 &= v_1 \cdot v_2 \\
        p_k &= p_{k-1} \cdot v_{k+1}, \quad \forall k \in \{2, \dots, |V_c|-2\} \\
        T_c &= p_{|V_c|-2} \cdot v_{|V_c|}
    \end{align}

    \item \textbf{Diferencia Absoluta ($-$):} Para $|V_c| = 2$:
    \begin{equation}
        |v_1 - v_2| = T_c \iff \text{AddAbsEquality}(T_c, v_1 - v_2)
    \end{equation}

    \item \textbf{Divisi\'on Reificada ($\div$):} Para $|V_c| = 2$, dado que el orden relativo de los operandos no est\'a predeterminado en la imagen, se formula mediante reificaci\'on introduciendo una variable booleana de direcci\'on $b_{\text{dir}} \in \{0, 1\}$ con \texttt{OnlyEnforceIf}:
    \begin{align}
        b_{\text{dir}} &\implies v_1 = T_c \cdot v_2 \label{eq:div_reif_1} \\
        \neg b_{\text{dir}} &\implies v_2 = T_c \cdot v_1 \label{eq:div_reif_2}
    \end{align}
\end{enumerate}

\subsection{Modelado de Jaulas: Variante B (Restricciones Globales de Tabla)}
En la Variante B, cada jaula $c$ se modela de forma extensional a trav\'es del cat\'alogo de tuplas factibles $\mathcal{T}_c \subset \{1, \dots, n\}^{|V_c|}$ calculado a priori mediante filtrado topol\'ogico y aritm\'etico:
\begin{equation}
    \text{AllowedAssignments}(V_c, \mathcal{T}_c)
    \label{eq:table_constraint}
\end{equation}
donde $\mathcal{T}_c$ descarta anticipadamente asignaciones con valores repetidos en celdas alineadas en la misma fila o columna de la grilla. Esto permite al solver mantener \textbf{Consistencia de Arco Generalizada (GAC)}~\cite{bessiere2006table} sobre la totalidad del \textit{scope} de la jaula sin requerir variables auxiliares intermedias.

\subsection{Variante C: Inferencia Conjunta Neuro-Simb\'olica v\'{\i}a Reificaci\'on}
\label{sec:joint_inference}
Para mitigar errores sensoriales de la etapa de visi\'on computacional~\cite{bradski2000opencv} en el reconocimiento \'optico de caracteres (OCR) (tales como confusi\'on entre glifos $8$ y $3$, o entre $+$ y $\times$), la Variante C formula un problema de optimizaci\'on combinatoria neuro-simb\'olica~\cite{deraedt2011constraint} de m\'axima verosimilitud consistente (MAP). Para cada jaula $c$, se extraen $K_c$ candidatos de lectura con su log-probabilidad normalizada $\log P_{c,k}$. Se introducen variables indicadoras de hip\'otesis:
\begin{equation}
    r_{c,k} \in \{0, 1\}, \quad \forall c, \; \forall k \in \{0, \dots, K_c - 1\}
\end{equation}
sujetas a la restricci\'on de partici\'on de selecci\'on \'unica:
\begin{equation}
    \text{ExactlyOne}(\{r_{c,0}, r_{c,1}, \dots, r_{c,K_c - 1}\}), \quad \forall c
\end{equation}
Cada hip\'otesis aritm\'etica se activa de manera condicional (reificada) v\'{\i}a \texttt{OnlyEnforceIf}:
\begin{equation}
    r_{c,k} \implies \mathcal{A}(V_c, T_{c,k}, \text{op}_{c,k})
\end{equation}
y se resuelve el objetivo de optimizaci\'on combinatoria:
\begin{equation}
    \max \sum_{c} \sum_{k=0}^{K_c - 1} \left\lfloor 1000 \cdot \log P_{c,k} \right\rfloor \cdot r_{c,k}
    \label{eq:map_objective}
\end{equation}
garantizando la selecci\'on de la hip\'otesis visual m\'as veros\'{\i}mil que admita soluci\'on l\'ogica estricta.

\section{An\'alisis de Complejidad y Evaluaci\'on Experimental del Solver}
\label{sec:complexity_and_experiments}

\subsection{Complejidad Te\'orica vs. Espacio de B\'usqueda Podado}
El espacio de b\'usqueda no restringido de una cuadr\'{\i}cula KenKen de orden $n$ es:
\begin{equation}
    |\mathcal{S}| = n^{n^2} = \mathcal{O}(n^{n^2})
\end{equation}
Para $n=3$, $|\mathcal{S}| = 3^9 = 19{,}683$. Para $n=9$, el espacio combinatorial alcanza $9^{81} \approx 1.96 \times 10^{77}$ estados, magnitud comparable al n\'umero estimado de \'atomos en el universo observable.

Bajo la inclusi\'on de las restricciones de Cuadrado Latino (\ref{eq:alldiff_row})--(\ref{eq:alldiff_col}), el espacio se reduce al orden del n\'umero de cuadrados latinos $L(n)$, acotado asint\'oticamente por $(n!)^{2n} / n^{n^2}$. Sin embargo, la propagaci\'on conjunta de CP-SAT~\cite{perron2024ortools} mediante \textit{Lazy Clause Generation} (LCG)~\cite{ohrimenko2009lazy} y propagaci\'on de l\'{\i}mites acota los dominios de forma tan agresiva que la soluci\'on es alcanzada en el \textbf{nodo ra\'{\i}z} (0 ramas exploradas en la pr\'actica para la mayor\'{\i}a de instancias).

\subsection{Discusi\'on de Hallazgos Emp\'{\i}ricos}
\begin{itemize}
    \item \textbf{Eficacia del filtrado GAC (Variante B):} En instancias de mayor complejidad ($n=9$), la Variante B resolvi\'o en $12.11\text{ ms}$ frente a $20.03\text{ ms}$ de la Variante A sin redundancia (reducci\'on del $39.5\%$). El filtrado GAC de jaula extensional poda valores inconsistentes desde la fase de preprocesamiento del solver.
    \item \textbf{Comportamiento de Restricciones Redundantes:} Para $n \le 6$, la inclusi\'on de (\ref{eq:redundant_sum}) a\~nade un sobrecosto marginal ($\approx 4$--$5\text{ ms}$) por verificaci\'on de sumas lineales. No obstante, para $n=9$, acelera la formulaci\'on aritm\'etica en un $35\%$ ($20.03\text{ ms} \to 13.03\text{ ms}$), confirmando que las restricciones implicadas resultan ventajosas cuando la combinatoria del problema crece.
    \item \textbf{Tolerancia a Ruido Visual:} La Variante C demostr\'o un $100\%$ de recuperaci\'on ante errores de lectura inyectados en glifos top-1, manteniendo tiempos de respuesta inferiores a $50\text{ ms}$.
\end{itemize}
